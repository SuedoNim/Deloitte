/* ============================================================
   FILE: components/ui.mdx
   Shared UI kit stub. Modern UI principles applied throughout:
   progressive disclosure (cards before tables), status color is
   never the only signal (chips carry text), keyboard-navigable
   tables, loading/empty/error states for every data region.
   ---------------------------------------------------------- */

export function StatCard({ label, value, delta, trend, caption }) {
  /* KPI card. delta: signed %; trend: "up"|"down"; caption: context. */
  return (
    <section className="stat-card" aria-label={label}>
      <h3 className="stat-card__label">{label}</h3>
      <p className="stat-card__value">{value}</p>
      {delta ? <p className={"stat-card__delta stat-card__delta--" + trend}>{delta}</p> : null}
      {caption ? <p className="stat-card__caption">{caption}</p> : null}
    </section>
  );
}

export function StateChip({ state, kind }) {
  /* kind: "healthy" | "warn" | "critical" | "neutral" — always shows state text (a11y). */
  return <span className={"chip chip--" + kind}>{state}</span>;
}

export function DataTable({ columns, rows, empty, loading, error, caption }) {
  if (loading) return <p role="status">Loading…</p>;
  if (error) return <p role="alert">Could not load data: {error}</p>;
  if (!rows || rows.length === 0) return <p>{empty || "No records."}</p>;
  return (
    <table className="data-table">
      <caption>{caption}</caption>
      <thead>
        <tr>{columns.map((c) => <th key={c.key} scope="col">{c.label}</th>)}</tr>
      </thead>
      <tbody>
        {rows.map((r, i) => (
          <tr key={r.id || i}>
            {columns.map((c) => <td key={c.key}>{c.render ? c.render(r) : r[c.key]}</td>)}
          </tr>
        ))}
      </tbody>
    </table>
  );
}

export function Callout({ title, tone, children }) {
  /* tone: "info" | "warn" | "unofficial". Used to flag unofficial/gap facts. */
  return (
    <aside className={"callout callout--" + tone} role="note">
      {title ? <h4>{title}</h4> : null}
      <div>{children}</div>
    </aside>
  );
}

export function Formula({ name, expr, where }) {
  return (
    <div className="formula">
      <p className="formula__name">{name}</p>
      <code className="formula__expr">{expr}</code>
      {where ? <p className="formula__where">{where}</p> : null}
    </div>
  );
}

export function StateTable({ transitions, caption }) {
  /* Renders a workflow state transition table (from → event/guard/action → to). */
  return (
    <table className="state-table">
      <caption>{caption}</caption>
      <thead>
        <tr><th>Current</th><th>Event (guard)</th><th>Action</th><th>Next</th></tr>
      </thead>
      <tbody>
        {transitions.map((t, i) => (
          <tr key={i}>
            <td>{t.current}</td>
            <td>{t.event}{t.guard ? " (guard: " + t.guard + ")" : ""}</td>
            <td>{t.action}</td>
            <td>{t.next}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}

export function MetricRadar({ dimensions, max }) {
  /* 0..max scores per dimension; prints a text list for screen readers. */
  return (
    <figure className="radar">
      <figcaption>Maturity by dimension (0–{max})</figcaption>
      <ul>
        {dimensions.map((d) => (
          <li key={d.name}>{d.name}: {d.score}/{max} {d.note ? "— " + d.note : ""}</li>
        ))}
      </ul>
    </figure>
  );
}
